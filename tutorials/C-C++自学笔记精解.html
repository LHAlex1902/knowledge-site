C240520.c最小完整程序 + 函数前置声明 
#include <stdio.h>          # 预处理指令：把标准输入输出库的头文件内容原样「贴」到此处
                                 # stdio = standard input/output；尖括号表示从系统目录查找

int f(int num);              # 【函数声明/原型】只告诉编译器「有这么个函数」，不写实现
                                 # 因为下面的 main 要用它，而它的定义在 main 后面 → 必须先声明
                                 # 结尾分号不能少：这是「声明」，不是「定义」

int main()                     # 程序入口，操作系统从这里开始执行
{                                # 函数体开始
    int x,y,z,sum=0;             # 一次声明 4 个 int 变量，sum 顺带初始化为 0
    scanf("%d",&x);              # %d 读一个十进制整数；&x 取 x 的地址 —— scanf 需要地址才能写入
                                 # 【易错】scanf 忘了写 & 是新手最常见的崩溃原因
    y=x+1; z=x+2;                # 一条语句里写两个赋值（分号分隔），合法但不推荐
    sum=f(x)+f(y)+f(z);         # 调用函数三次，把三个阶乘结果相加
    printf("%d %d %d %d",sum,x,y,z);   # 按顺序把 4 个 int 填进 4 个 %d 占位符
}                                # main 没写 return 0，C99 标准下默认返回 0（表示成功）

int f(int num)                  # 【函数定义】这里才是真正的实现
{
    int i,t=1;                   # t 是累乘器，必须初始化为 1（初始化为 0 的话乘出来永远是 0）
    for (int i = 1; i <= num; i++)   # 循环变量 i 从 1 到 num（<= 所以包含 num）
        t=t*i;                   # 累乘。注意这里用的是内层新声明的 i
    return t;                    # 把结果返回给调用者
}                                # 【可优化】外层那个 int i 从未被使用 —— 属于冗余变量，删掉更干净
⚠️ 变量遮蔽（Shadowing）：函数里外层写了 int i，循环里又写了 int i —— 内层的 i 会「盖住」外层的 i，外层的 i 从声明后就再没人用。 能编译，但会让读代码的人困惑。

6.1 递归三要素

C2024年5月24日.c递归求阶乘 
long x(long n)                    # 返回 long，参数也是 long（想支持更大的数）
{
    if (n == 1)                   # 【终止条件】递归必须有一个不再调用自己的出口
        return 1;                  # 1! = 1，递归「触底」
    else
        return n*x(n-1);            # 【递归调用】n! = n × (n-1)!，参数在变小，最终会碰到 n==1
                                 # 展开（n=4）：4*x(3) → 4*3*x(2) → 4*3*2*x(1) → 24
}
int main()
{
    int y;
    scanf("%d",&y);
    printf("%ld",x(y));           # %ld 对应 long（写成 %d 在 64 位上会输出错乱）
}                                # 【隐患①】输入 0 或负数则永远碰不到 n==1 → 无限递归 → 栈溢出崩溃
                                 # 【隐患②】Windows 下 long 是 32 位，13! 就溢出 → 应该用 long long

6.3 三数最大值 —— 一个隐藏的逻辑漏洞

C2024年5月24日test1.c嵌套 if 的坑 
int bj(int a,int b,int c)
{
    if (a > b)                       # 外层判断：a 是否大于 b
        if (b > c)                    # 内层判断
            return a;                 # a > b 且 b > c → a 最大 ✓
        else if (c > a)
            return c;                 # a > b 且 c > a → c 最大 ✓
                                 # 🚨 【漏洞】当 a > b 且 b <= c 且 c <= a（即 a ≥ c > b）时，
                                 # 两个分支都不满足 → 直接掉到最后的 return c —— 但最大值是 a！
                                 # 反例：a=5, b=1, c=3 → 返回 3，实际最大是 5 ❌
    else if (b > c)                   # 【缩进误导】这个 else 属于最外层的 if (a > b)
                                 # 但缩进看起来像属于内层 if，读代码的人极易误判
        return b;                     # a ≤ b 且 b > c → b 最大 ✓
    return c;                        # 兜底：a ≤ b 且 b ≤ c → c 最大 ✓
}

// 【推荐改法】一行搞定，没有嵌套歧义：
// int m = a; if (b > m) m = b; if (c > m) m = c; return m;
// 或者：return a > b ? (a > c ? a : c) : (b > c ? b : c);
⚠️ 「悬空 else」的教训：嵌套 if 不加 {} 时， else 会绑定到最近的、还没配对的那个 if。 这个例子里语法配对其实是对的，但缩进撒了谎，让人误以为配错了。 结论：嵌套 if 一定要加 {}，永远不要赌自己的缩进。

7 🎯 指针与数组

指针的本质只有一句话：指针是一个存「地址」的变量。 &变量 取地址，*指针 取出地址上的值。

7.1 指针遍历数组

C2024年5月27日.c正序输入、逆序输出 
int main()
{
    int a[10];                       # 10 个 int 的数组，在栈上连续排列
    int *q=a;                       # 【关键】数组名 a 会自动「退化」成首元素地址
                                 # 等价于 int *q = &a[0];
    for ( ; q < a+10; q++)        # 【写法】三个表达式里第一个留空，因为 q 已在外面初始化
                                 # a+10 是「首地址 + 10 个 int 的长度」= 数组末尾的再后一位
                                 # q++ 让指针前进 sizeof(int) 个字节（不是前进 1 字节！）
        scanf("%d",q);               # q 本身就是地址，所以不用写 &q（等价于 &a[i]）
    for (q=a+9 ; q >= a; q--)       # 【指针倒着走】q 指向最后一个元素，每次减 1
        printf(",%d,",*q);             # *q 解引用取出当前值 → 实现逆序输出
}                                # 【知识点】q >= a 这个比较在数组范围内是安全的，能正确终止循环 ✓
写法含义
a / &a[0]首元素地址
a + i / &a[i]第 i 个元素的地址
*(a + i)第 i 个元素的值（等价于 a[i]）
a + 10越界的「哨兵地址」，只能比较、不能解引用
q++指针前进一个元素（不是 1 个字节）

7.2 指针交换 —— 为什么必须传地址

{

    scanf("%d%d",&a,&b);
    jh(&a,&b);                      # 【关键】传的是 &a、&b（地址），不是 a、b（值）
                                 # 写成 jh(a, b) 会编译报错（int 不能当 int*）
    printf("%d %d",a,b);             # 输出验证：已交换
}
「值传递」vs「地址传递」：
C 语言的参数永远是值传递 —— 函数收到的是实参的副本。
· 传 a（值）→ 函数改的是副本，原变量不动。
· 传 &a（地址）→ 副本是地址，但它指向的还是原来那块内存，所以能改到原变量。
这就是为什么 scanf 也必须写 &。

7.3 整数除法陷阱

C2024年7月1日-1.c求和与求平均 
double avg(int a,int b)
{
    double avg;
    avg=a/b;                         # 🚨 【重大陷阱】a、b 都是 int → 先做整数除法（丢掉小数）再赋给 double
                                 # 例：a=7, b=2 → 7/2 = 3（不是 3.5）→ avg = 3.00 ❌
                                 # 正确写法：avg = (double)a / b;  ← 先把 a 提升为 double，除法才按小数算
    printf("%.2f\n",avg);          # 打印出来看着像正常两位小数，所以很难发现错了
    return avg;
}

int sum(int sz[],int x)          # 【知识点】int sz[] 里的 [] 是摆设，实际收到的是 int* 指针
                                 # 所以 sizeof(sz) 得到的是指针大小（4/8），不是数组总字节数！
                                 # 这就是为什么必须额外传一个长度参数 x
{
    int sum=0;                     # 【隐患】局部变量名 sum 与函数名 sum 同名
                                 # 局部变量会遮蔽函数名，能编译但读起来很绕 → 建议叫 total
    for (int i = 0; i < x; i++)
        sum+=sz[i];                  # 累加，等价于 sum = sum + sz[i]
    printf("%d\n",sum);
    return sum;
}

int main()
{
    int y[]={1,2,3};               # 由初始值自动推断长度为 3
    double a;
    a=avg(sum(y,3),3);              # 【嵌套调用】先算 sum(y,3)=6，再传给 avg(6, 3)
                                 # 6/3 = 2 恰好整除，所以这次结果 2.00 看起来正常 → 坑被掩盖了
                                 # 换成 {1,2,4}（和=7）就会得 2.00 而不是 2.33
    printf("%.2f\n",a);
}
🚨 整数除法 —— C 语言最经典的坑之一： 两个整数相除，结果永远是整数（小数部分直接截断，不是四舍五入）。 只要有一边是浮点类型，除法才会按小数算。所以在算平均值、百分比、比率时， 一定要检查是不是 int / int。

7.4 手写字符串长度与回文判断

C2024年7月2日.c字符串处理的本质 
int zfc(char n[])                  # 求「最大下标」（zfc = 字符串的拼音缩写）
{
    int i=0;
    while (n[i] != '\0')               # 【关键】C 字符串靠 '\0'（空字符）标记结尾
                                 # 所以「字符串长度」≠「数组长度」，要一直数到 '\0' 才停
        i++;                         # i 不断自增，最终停在 '\0' 的位置上
    i--;                             # 退一格 → 得到「最后一个有效字符」的下标
    return i;                        # 注意：返回的不是长度，是「长度 - 1」
                                 # 所以 main 里要写 zfc(s)+1 才是真正的字符个数
                                 # 【建议】函数名改成 max_index 更准确，否则容易误用
}

int hs(char n[],int b)                # b 接收 zfc 的返回值（最大下标）
{
    int i = 0;
    for (; i < b/2; i++)               # 只比较前半段与后半段的对称位置
    {
      if (n[i] != n[b-i])             # 对称位置：下标 i ↔ 下标 b-i
                                 # 例 "abcba"：b=4，比 n[0]vs n[4]、n[1]vs n[3] ✓
      {
        i=-1;                       # 发现不相等 → 用 -1 作为「不是回文」的标记
        break;                      # 提前结束，不用再比了
      }
    }
    return i;                        # -1 = 不是回文；正数 = 是回文
}

int main()
{
    char s[99];
    int x;
    printf("输入字符串\n");
    scanf("%s",s);                  # 【注意】%s 遇到空格就停，不能输入带空格的句子
    printf("字符个数：%d\n",zfc(s)+1);   # 最大下标 + 1 = 字符个数
    x=hs(s,zfc(s));
    if (x > 0)
        printf("此字符串是回文\n");
    else if (x < 0)
        printf("此字符串不是回文\n");
                                 # 【漏洞】x == 0 时两个分支都不进，什么都不打印
                                 # 什么时候 x==0？单字符输入（如 "a"）时 b=0，循环不执行
}
⚠️ 这个回文函数对「偶数长度」会漏检： 循环条件是 i < b/2，而 b 是最大下标（长度-1）。
· 长度 5（b=4）：b/2=2，i 取 0、1 → 比较了 2 对 ✓ 完整
· 长度 4（b=3）：b/2=1，i 只取 0 → 只比了 1 对，中间那对 n[1]vs n[2] 没比 ❌
反例："abca" → 只比 n[0]vs n[3]（都是 'a'）就通过，误判为回文。
修正：条件改为 i <= b/2 或 i < (b+1)/2。

8 ⚙️ 宏定义 —— 文本替换的威力与陷阱

#define 是预处理指令：在真正编译之前， 预处理器会把代码里的宏名原样替换成宏体。 它不理解类型、不检查语法，纯粹是「查找-替换」。

8.1 基础用法

CDEFINE-1.c三种宏 + 一个对比用的函数 
void HH(int *a,int *b)              # 普通函数：指针交换（对比下面的宏 —— 函数有类型检查）
{
    int temp;
    temp=*a;
    *a=*b;
    *b=temp;
}

#define LF(x) x*x                    # 宏①：求平方。⚠️ 参数和整体都缺括号
                                 # 🚨 LF(2+3) 展开成 2+3*2+3 = 11（不是 25）
                                 # 正确写法：#define LF(x) ((x)*(x))

#define CY(x) x%4                    # 宏②：对 4 取余。同样缺括号
                                 # CY(a+1) 展开成 a+1%4 → % 优先级高于 + → 结果错
                                 # 正确写法：#define CY(x) ((x)%4)

#define X1(x,y) ((x*y)>100?1:0)      # 宏③：条件表达式。这个写得比较规范 ✓
                                 # 整体加了括号；但内部 x、y 仍建议各自加括号防意外

int main()
{
    int x,y;
    scanf("%d%d",&x,&y);
    printf("a=%d\n",LF(x));          # 展开后：printf("a=%d\n", x*x); —— 传单个变量时碰巧正确
    printf("b=%d\n",CY(x));          # 同上，x%4 碰巧正确
    printf("c=%d\n",X1(x,y));        # 展开后：((x*y)>100?1:0) ✓
}
🚨 宏的第一铁律：所有参数和整体都要加括号。
#define LF(x) x*x 写起来最自然，但 LF(2+3) 会得到 11； 写成 #define LF(x) ((x)*(x)) 才能保证任何情况下都对。
因为宏是文本替换，不是函数调用 —— 它不会先把参数算出来再代入。

8.2 交换宏 —— 一个逻辑错误

                                 # ⚠️ 隐患：若写成 if (x) HH(m,n); else ... 会展开成
                                 # if (x) {...}; else ... → 分号断开，else 找不到 if → 编译错误
                                 # 规范写法：#define HH(a,b) do{int c=a;a=b;b=c;}while(0)

#define HH2(a,b) a=b                   # 单纯赋值宏

int main()
{
    int x,y,z;
    scanf("%d%d",&x,&y);
    // printf("%d%d",HH2(y,x),HH2(x,y));   # 已注释：本想交换，但这样写结果不可预期
    // HH(x,y,z);                  # 已注释：HH 只接收两个参数，传三个会报错
    printf("q%d%d\n",x,y);        # 打印交换前的值
    z=HH2(z,x);                    # 展开后：z = z = x;  ← 连续赋值
                                 # 效果是「z 和 x 都变成 x 的值」，z 原本的值被覆盖丢失了！
                                 # ❌ 这不是交换，是逻辑错误
    x=HH2(x,y);                    # 展开后：x = x = y;  → x 变成 y
    y=HH2(y,z);                    # 展开后：y = y = z;  → y 变成 z（而 z 已等于 x 了）

}

// 【正确做法】交换必须借助临时变量：
// int temp = x; x = y; y = temp;   ← 或者直接调用 DEFINE-1.c 里那个指针版 HH 函数

8.3 跨文件包含 —— 不规范但理解了机制

CDEFINE-2.c#include 一个 .c 文件

                                 # 所以 DEFINE-2.c 里能用 DEFINE-1.c 的 HH 函数 ✓
                                 # 🚨 【问题】但 DEFINE-1.c 里也有一个 main()！两个 main 一起进来 → 冲突

{
    int x,y;
    scanf("%d%d",&x,&y);
    printf("%d-%d\n",x,y);        # 打印交换前
    HH(&x,&y);                      # 调用从 DEFINE-1.c 借来的指针交换函数 ✓
    printf("%d-%d\n",x,y);        # 打印交换后 —— 真的交换成功（对比宏版）

✅ 现代建议：能用 const 常量就别用 #define 常量； 能用函数（或 C++ 的 inline）就别用函数式宏。 宏保留给「条件编译」（#ifdef 调试开关）这类真正需要它的场景。

#include <malloc.h>              # malloc/free 需要的头文件（标准是 <stdlib.h>，你两个都包含了）

struct XX                          # 学生信息结构体
{
    int id;
    char name[10];
    int age;
    char address[20];
};

struct XX input()                  # 读入一条信息并返回整个结构体
{
    struct XX x;
    scanf("%d %s %d %s",
    &x.id,
    x.name,                      # 数组不用 &
    &x.age,
    x.address);
    return x;                        # 【知识点】返回值会整体拷贝一次（现代编译器多会优化掉）
}

int main()
{
    int x;
    printf("学生数\n");
    scanf("%d",&x);
    struct XX xs[x];                   # 【VLA】变长数组建在「栈」上
                                 # 如果 x 输入 100000，约需 3.8MB → 栈通常只有 1~8MB → 立即栈溢出崩溃
    struct XX *p=&xs[0];              # 指针指向数组首元素
    p=(struct XX *)malloc(x * sizeof(struct XX));
                                 # 【malloc 三件套】
                                 #   x * sizeof(...)  → 算出总共需要多少字节
                                 #   malloc(...)      → 在「堆」上申请，返回 void*
                                 #   (struct XX *)    → 强制类型转换成需要的指针类型
                                 # 🚨 【严重问题】这块 malloc 出来的内存和上面的数组 xs 毫无关系！
                                 # 后面读写数据全用的是 xs[...]，p 除了被 free 之外完全没被使用 → 纯属多余
    if (p == NULL)                     # ✓ 检查申请是否成功（内存不足时 malloc 返回 NULL）
    {
        printf("malloc error\n");
        exit(1);
    }
    printf("malloc success\n");
    for (int i = 0; i < x ; i++)
    {
        printf("学生信息\n");
        xs[i]=input();                # 函数返回结构体，直接赋给数组元素 ✓
    }

    printf("%d %s %d %s",xs[1].id,...);   # 【隐患】固定取下标 1，如果只输入了 1 个人就访问越界

    FILE *fp;
    if (fopen_s(&fp,"C:\\FILE\\record2.txt","w+") != 0)
    {
        printf("File open fail\n");
        free(p);                        # 出错路径上也记得释放内存 ✓ 好习惯
        exit(1);
    }
    for (int i = 0; i < x; i++)
    {
        fwrite(&xs[i],sizeof(struct XX),1,fp);
                                 # 【知识点】sizeof(struct XX) 包含编译器插入的「对齐填充字节」
                                 # 例：int(4) + char[10] + int(4) 实际可能是 20 字节而不是 18
                                 # 所以文件会比「数据实际大小」大一点点，这是正常的
        // fwrite(&(p+x)->name,10,1,fp);   # 已注释：想按字段拆开写（p+x 还是错的，应该是 p+i）
    }
    free(p);                          # ✓ 有 malloc 就必须有 free，否则内存泄漏
    fclose(fp);                       # ✓ 文件用完要关
}
内存区域存什么生命周期大小限制
栈 Stack局部变量、函数参数、VLA 数组函数返回就销毁小（1~8MB）→ 大数组会溢出
堆 Heapmalloc 申请的内存手动 free 才释放大（受物理内存限制）
全局/静态区全局变量、static 变量整个程序运行期中等，且默认清零
⚠️ 这份代码暴露的设计问题： 既然用 malloc 在堆上开了空间存学生数据，就应该全程用 p 访问 （p[i] 和 xs[i] 语法一样），把 VLA 那行删掉。 两个都写，结果就是「申请了内存却不用」—— 白占堆空间，还容易让人误判数据存在哪。
malloc 使用四步口诀： ① 算大小 n * sizeof(类型) → ② 申请 malloc(...) → ③ 检查 if (p == NULL) → ④ 用完 free(p) 并置空 p = NULL。

11.3 二进制读写 —— 一个越界 bug

C2024年7月19日.c / test.cfwrite + rewind + fread

                                 # 正确写法：fwrite(test, sizeof(test), 1, fp);   ← 写 1 个 8 字节
                                 # 或更明确：fwrite(test, sizeof(char), 8, fp);

🚨 这是本次整理里最严重的一个 bug： fwrite(test, sizeof(test), 2, fp) 要写 16 字节，但 test 只有 8 字节。
第 2 个「元素」读的是数组后面的内存 —— 属于未定义行为。 虽然通常只是把垃圾数据写进文件（不会立即崩溃），但在某些情况下会触发段错误， 而且写出的文件内容「不可复现」。
根本原因：把「元素大小」和「元素个数」搞混了。只想写一个字符串时，个数应该填 1。
fwrite / fread 参数对照

//      │     │      │     └─ 文件指针
//      │     │      └─────── 要写「几个」元素
//      │     └────────────── 每个元素「多少字节」
//      └──────────────────── 数据从哪里读（地址）
//
// 实际写入字节数 = size × count
// 所以必须保证：size × count ≤ 数据实际占用的字节数

// 常见三种正确写法：
fwrite(str, sizeof(char), 8, fp);          // 写 8 个字节
fwrite(str, sizeof(str), 1, fp);           // 写整个数组（1 个元素）
fwrite(arr, sizeof(struct XX), n, fp);       // 写 n 条结构体记录 ✓ 你写文件用的就是这个

"r"只读❌ 失败——开头
"w"只写✅ 新建⚠️ 清空全部内容开头
"a"追加写✅ 新建在后面接着写末尾
"r+"读写❌ 失败——开头

                                 # 自动选中第一个版本 ✓

void getpoint (long& x,long& y)        # 【引用参数】直接修改调用者的变量，不用写指针
{
    HWND hWnd = GetForegroundWindow();   # 拿到「当前最前面的那个窗口」的句柄
                                 # HWND = window handle，Windows 里每个窗口的身份证
    POINT dian;                       # POINT 是系统定义的结构体，有两个成员 x、y
    GetCursorPos(&dian);              # 获取鼠标当前坐标 —— 这是「屏幕坐标」（相对整个屏幕）
    ScreenToClient(hWnd,&dian);       # 把「屏幕坐标」换算成「窗口客户区坐标」
                                 # 换算后 (0,0) 变成窗口客户区的左上角，方便记录相对位置
    x = dian.x;                       # 通过引用把结果写回调用者的变量 ✓
    y = dian.y;
}

                                 # 合在一起 = 一次完整的左键单击
                                 # 后两个参数 0,0 是「滚轮/附加数据」，单击时为 0
}
