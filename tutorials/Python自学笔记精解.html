3 🎀 装饰器 — 不修改原函数，给它「套一层壳」

核心思想：Python 里函数也是对象，可以被当作参数传来传去。 装饰器就是「接收一个函数，返回一个新函数」的函数，用 @装饰器名 贴在目标函数上面。

3.1 基础.py 里的 time_decorator

Python基础.py自定义计时装饰器 
# 无参数装饰器：
def time_decorator(func):               # 装饰器本身：func 参数就是「被装饰的那个函数」
    def wrapper(*any_what, **any_type):  # 内层函数：它将「顶替」原函数被执行
                                          # *any_what 收所有位置参数成元组；**any_type 收所有关键字参数成字典
                                          # 这样写的好处：不用管原函数有几个参数，一个壳能套所有函数
        start_time = time.time()          # 记录开始时间（从1970年至今的秒数，浮点）
        print(f"开始执行：{func.__name__}")  # __name__ 是函数自带属性，拿到函数名字符串
        print("-" * 90)                   # 字符串乘法 = 重复 90 次，打印一条分割线

        old_func = func(*any_what, **any_type)   # 【关键】这里才真正执行原函数
                                          # 星号的作用是把收到的参数「拆开」再传进去
                                          # 返回值存到 old_func，后面要原样还给调用者

        end_time = time.time()            # 记录结束时间
        print("-" * 90)
        print(f"{func.__name__}执行结束。")
        # print(f"任务入参：{any_what}{any_type}")   # 已注释：想打印入参但太啰嗦
        print(f"耗时： {end_time - start_time:0.2f}秒")   # :0.2f = 保留2位小数的浮点格式

        t = time.localtime()              # 把当前时间戳转成「时间结构体」（年月日时分秒）
        print("当前时间：", time.strftime("%Y-%m-%d %H:%M:%S", t))   # strftime 按格式串把时间转成可读字符串

        return old_func                  # 【关键】把原函数的返回值原样交还，否则外部拿到的是 None

    return wrapper                       # 【关键】装饰器返回内层函数（注意：不加括号，返回的是函数本身）

3.3 self_use.py 里的 eva_time（另一版计时装饰器）

Pythonself_use.pyeva_time — 更精简的计时装饰器 
# 计算函数执行时间
def eva_time(fun):                                  # 参数名换成 fun，和上面 func 是一回事
    def time_wrapper(*args, **kwargs):                 # 收任意参数
        print("当前执行函数：", fun.__name__)              # 用逗号分隔输出，自动加空格
        print('》', "-" * 50, '《')                      # 装饰性分隔线
        start_time = time.time()                        # 开始计时
        result = fun(*args, **kwargs)                   # 执行原函数，拿到结果
        end_time = time.time()                          # 结束计时
        print('》', "-" * 50, '《')
        print(f"函数：{fun.__name__}执行完毕，"
              f"参数1：{args}, 参数2：{kwargs}，"      # 相邻字符串会自动拼接成一条
              f"耗时{end_time - start_time:0.2f}秒。")
        return result                                   # 返回原函数结果
    return time_wrapper                                 # 返回新函数

# 下面这两个是「空壳装饰器」，原封不动返回结果，没做任何加工：
def gui(fun):
    def gui_wrapper(*args, **kwargs):
        result = fun(*args, **kwargs)                   # 只是调一下原函数
        return result
    return gui_wrapper

def demo(fun):                                      # 与 gui 完全一样（复制粘贴留下的）
    def gui_wrapper(*args, **kwargs):                 # 内层函数也叫 gui_wrapper，但作用域独立不冲突
        result = fun(*args, **kwargs)
        return result
    return gui_wrapper
💡 观察：eva_time 和 time_decorator 是同一个思路的两次练习 —— 这正是学装饰器最有效的方式：自己写两遍就记住了。 而 gui() / demo() 是「还没想好要加工什么」的占位模板。

    print(r"1、{} \n 2、{}".format(c, d))   # r 只影响「字面量」的转义，不影响 .format()
                                        # 所以这里 {} 依然会被 c、d 替换，而 \n 会原样打印成两个字符

    # 格式化输出
    for i in range(2, 11):              # range(2,11) → 2 到 10（含2不含11）
        print("{0:5d}|{1:5d}|{2:5d}".format(i, i*i, i*i*i))
                                        # {0:5d}：第0个参数，宽度5右对齐的整数；i²、i³ 同理
                                        # 效果：输出 i、i的平方、i的立方 三列表格
格式化方式写法特点
① str.format()"a={}".format(a)占位符 {}，可加索引和格式说明
② 逗号分隔print("a=", a)最简单，自动加空格，但不能精细控制格式
③ f-stringf"a={a}"最推荐，可读性最好，还能写表达式
④ % 格式化"a=%d" % a老写法（C 语言风格），新代码少用
⚠️ 常见的坑：input() 返回的永远是字符串。 所以要做数学运算必须先 int() 转换 —— 你后面代码里到处出现的 int(input(...))、map(int, ...) 就是在处理这个问题。

@time_decorator
def if_for():
    list_for = list(map(int, (input('输入一组整数，空格区分').split())))
                # 从里往外拆解这行「三连套」：
                #   ① input().split()  → ['1','2','3'] 字符串列表
                #   ② map(int, ...)    → 对每个元素套用 int()，得到 map 对象（惰性，不是列表）
                #   ③ list(...)        → 把 map 对象转成真正的列表 [1, 2, 3]
    tup_for = tuple(list_for)            # 列表 → 元组（类型转换）
    set_for = set(list_for)              # 列表 → 集合（顺手去重）
    dict_for = {'2': 2, '3': 3, '4': 4, '5': 5}   # 直接写字典字面量

    tub_all = [list_for, tup_for, set_for, dict_for]   # 列表里装容器：嵌套结构
    print(tub_all)                      # 打印这个「容器大集合」
    print(type(tub_all))                # type() 查看类型，结果是 <class 'list'>

    for i in tub_all:                    # 遍历列表 → i 依次是 列表、元组、集合、字典
        print(i)
    for i in list_for:                  # 遍历普通列表 → i 依次是每个数字
        print(i)

    # range()
    for i in range(1, 50, 2):            # range(起点, 终点不含, 步长) → 1,3,5,...,49 所有奇数
        print(i)
    list_range = list(range(1, 10))       # range 对象转列表 → [1,2,...,9]
    print(list_range)

    # range 配合 for 将列表 创建为字典
    dict_range = dict()                    # 先建一个空字典
    for i in range(len(list_range)):   # i 从 0 到 8，正好是列表的合法下标
        dict_range[i] = list_range[i]    # 用下标当键、元素当值，逐个塞进字典
    print(dict_range)                   # → {0:1, 1:2, ..., 8:9}

7 ✨ 推导式 — 一行生成容器

推导式（Comprehension）是 Python 的招牌特性：把「循环 + 判断 + 追加」三件事压缩成一行。 你的注释写得很到位：「方便快捷生成列表[]、元组()、集合{}、字典{}」。
Python基础.py → tds()四种推导式对照 
@eva_time
def tds():
    # 列表：生成一个1到10的偶数列表
    list_tds = [x for x in range(10) if x % 2 == 0]
                # 读法：「把 x 放进列表，其中 x 来自 range(10)，且 x 是偶数」
                # 等价于传统写法：result=[]; for x in range(10): if x%2==0: result.append(x)
    print("列表推导式：", list_tds)      # → [0, 2, 4, 6, 8]

    # 元组：与列表区别在于使用()，且推导式返回的是生成器对象，需要tuple()转换
    tup_tds_scqdx = (x for x in range(10) if x % 2 == 0)
                # 【重要】用 () 包起来的不是「元组推导式」，而是「生成器表达式」！
                # 它不会立刻算出所有值，而是等你取值时才算（惰性求值）
    print("生成器对象，需要转换：\n", tup_tds_scqdx)   # 打印出来是 <generator object ...> 而不是数据
    tup_tds = tuple(tup_tds_scqdx)       # 用 tuple() 把它「跑一遍」转成真元组
    print("元组推导式：", tup_tds)

    # 集合: 同列表
    set_tds = {x for x in range(10) if x % 2 == 0}

8 ⛓️ 迭代器与生成器 — 一个「随用随取」的流水线

先记住两句话：
· 迭代器 = 能记住「读到哪儿了」的对象，用 next() 取下一个。
· 生成器 = 用 yield 写的函数，自动具备迭代器能力，而且是「用一次算一次」。

8.1 生成器函数 yield_fun

Python基础.py → yield_fun()yield 把函数变成生成器 
# 生成器函数，给下文使用
# 此处举例 计算1到n乘以2
@time_decorator                 # 【注意】给生成器函数套装饰器，耗时统计其实不准（见下方提示）
def yield_fun(n):
    a, b = 1, 2                    # 元组解包：a=1, b=2 一行赋两个值
    while a < n:                     # 条件循环，a 从 1 涨到 n-1
        print("a的数值及a*2的值为:", a, end='，')   # end='，' 把默认的换行符换成逗号，输出接在同一行
        yield a * b                  # 【核心】yield = 产出值后「暂停」，函数状态被冻结
                                     # 下次调用 next() 时从这一行往下继续执行
        a += 1                       # 恢复执行后把 a 加 1，再回头判断 while
⚠️ 你这份代码里一个隐藏的知识点： yield_fun 被 @time_decorator 装饰后，调用它得到的是 生成器对象，而不是执行结果 —— 因为生成器函数一被调用就立刻返回、函数体一行都不跑。 所以装饰器里打印的「耗时 0.00 秒」统计的只是创建生成器的耗时， 真正干活的时间没被算进去。这是「装饰器 + 生成器」组合时的经典陷阱。

8.2 迭代器 iter / next

Python基础.py → iter_next()iter() 与 next() 的配合 
@time_decorator
def iter_next():
    # 迭代器的iter() next(),生成器的yield yield from 将函数转变为生成器函数

    # 迭代器部分
    xx = int(input("迭代次数5,<5则报错，>=5则for循环接力输出，输入列表长度:"))
                                     # 转成整数后作为列表长度（这行是判断边界用的实验）
    ll = [x for x in range(xx)]        # 列表推导式 → [0, 1, 2, ..., xx-1]，是可迭代对象
    print(ll)
    ll_iter = iter(ll)               # 【关键】iter() 把「可迭代对象」转成「迭代器对象」
                                     # 区别：列表本身没有「读到哪了」的状态，迭代器有
    print(next(ll_iter), end=' ')    # next() 取出下一个值，指针自动后移
    print(next(ll_iter), end=' ')
    print(next(ll_iter), end=' ')
    print(next(ll_iter), end=' ')
    print(next(ll_iter))            # 迭代次数超过则弹报错StopIteration
                                     # 这是你自己验证出的结论：取完还取 → StopIteration 异常

    print("for循环调用next自动输出，不弹报错")
    for i in ll_iter:                # for 会不断调用 next()，取完后「自动捕获 StopIteration 并正常结束」
        print(i, end=' ')              # 因为前面已经取走 5 个，所以这里从第 6 个开始接续输出

    print("\n列表ll的类型为：", type(ll))            # → list，可迭代对象
    print("迭代器对象ll_iter的类型为：", type(ll_iter))   # → list_iterator，真正的迭代器

    # 生成器 使用了关键字yield的函数，被称为生成器
    # 此处举例 计算1到n乘以2
    yield_n = yield_fun(int(input("\n生成器部分：\n设置总数：")))
                                     # 调用生成器函数 → 立刻返回生成器对象，函数体还没跑
    print(next(yield_n), end=' ')    # 第一次 next() 才真正开始执行到第一个 yield，产出 1*2=2
                                     # 此处调用一次，不论n设置多少，仅输出一次 ← 你的注释总结得很准确

    # 自由控制
    for i in range(int(input("\n输入小于n-1的数来继续输出循环："))):
        try:                            # try/except：捕获可能出现的异常
            print(next(yield_n))          # 手动取下一个值
        except StopIteration:           # 超出n值时报错控制 ← 取完了就跳出循环，而不是崩掉
            break

    # 全部输出
    print("\nfor循环补足n里余下的次数")
    for i in yield_n:               # 直接 for 遍历生成器，取完自动结束不再报错
        print(i)
    return                          # 显式 return（其实可省略，函数末尾默认返回 None）
✅ 你已经搞懂的核心区别： for 循环内部会自动处理 StopIteration，所以不会报错； 而手写 next() 一旦取完，就会抛异常，必须自己 try/except 或干脆改用 for。

· reduce(lambda x,y: x+y, lam) → sum(lam)

    def print_2(self):
        print(f"身高：{self.height}cm")
        print(f"体重：{self.weight}kg")
        print(f"BMI: {self.weight / ((self.height * self.height) / 10000):.2f}")
                                     # BMI 公式 = 体重kg ÷ 身高m²
                                     # 因为 height 输入的是 cm，所以 ÷10000 变成 m²（100²=10000）
                                     # :.2f 保留 2 位小数
        a = self.weight / ((self.height * self.height) / 10000)   # 把 BMI 存到变量 a 用于分级判断
        if a < 18.5:
            print("你的BIM结果是：偏瘦")   # 小笔误：BIM 应为 BMI
        elif 18.5 <= a < 24.9:             # Python 独有：可以写「链式比较」
            print("你的BIM结果是：正常")
        elif 25 <= a < 29.9:
            print("你的BIM结果是：超重")
        elif a > 30:
            print("你的BIM结果是：肥胖")
        else:
            print("无")                    # ← 区间有缝隙：24.9~25、29.9~30 之间会落到这里

    def print_3(self, a):                # a 是「目标体重」
        b = self.weight - a              # 当前体重 - 目标体重
        if b >= 0:                        # 正数说明要减
            print(f"还需要减{self.weight - a:0.2f}kg")
        else:                            # 负数说明偏瘦要增
            print(f"需要增肥{a - self.weight:0.2f}kg")   # 取绝对值来表达「增多少」
